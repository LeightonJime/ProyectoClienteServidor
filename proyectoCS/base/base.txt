--------------CREAR TABLA USUARIOS-------------------

use biblioheredia;
create table usuarios(
    id_usuario int auto_increment primary key,
    username varchar(100) not null,
    email varchar(100) not null,
    password varchar(255) not null
);

--------------CREAR TABLA LIBROS---------------

use biblioheredia;
create table libro(
    isbn varchar (16) not null primary key, --el isbn viene siendo el identificador
    titulo varchar (100) not null,
    escritor varchar (100),
    anno varchar (16),
    descr varchar (255)
);

--------------CREAR COLECCIONES---------------

use biblioheredia;
create table coleccion(
   id_coleccion int auto_increment primary key,
   c_nombre varchar (100) not null
);

use biblioheredia;
create table lista(
    id_lista int auto_increment primary key,
    fk_id_coleccion int, 
    fk_isbn varchar (16),
    foreign key (fk_id_coleccion) references coleccion(id_coleccion),
    foreign key (fk_isbn) references libro(isbn)
);

--------------CREAR PRÉSTAMOS---------------

use biblioheredia;
create table prestamo(
    id_prestamo int auto_increment primary key,
    fk_id_usuario int,
    fk_isbn varchar (16),
    fecha_prestamo date,
    fecha_vence date,
    foreign key (fk_id_usuario) references usuarios(id_usuario),
    foreign key (fk_isbn) references libro(isbn)
);

----------------RATING LIBROS------------
USE biblioheredia;

CREATE TABLE rating_libro (
    id_rating INT AUTO_INCREMENT PRIMARY KEY,
    fk_id_usuario INT NOT NULL,
    fk_isbn VARCHAR(16) NOT NULL,
    estrellas TINYINT NOT NULL,
    resena VARCHAR(500),
    fecha_creacion DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uk_usuario_libro (fk_id_usuario, fk_isbn),
    FOREIGN KEY (fk_id_usuario) REFERENCES usuarios(id_usuario),
    FOREIGN KEY (fk_isbn) REFERENCES libro(isbn)
);
